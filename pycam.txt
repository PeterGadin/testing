from picamera2 import Picamera2
import cv2
import time

picam2 = Picamera2()

# Request RGB888 format from the camera
config = picam2.create_preview_configuration(main={"size": (640, 480), "format": "RGB888"})
picam2.configure(config)
picam2.start()

# Warm-up delay to let Auto Exposure settle
time.sleep(1)

print("Camera running with corrected colors. Press 'q' to exit.")

try:
    while True:
        # Capture raw frame (RGB)
        frame_rgb = picam2.capture_array()

        # SWAP CHANNELS: Convert RGB to BGR for OpenCV
        frame_bgr = cv2.cvtColor(frame_rgb, cv2.COLOR_RGB2BGR)

        # Display corrected frame
        cv2.imshow("Adjusted Preview", frame_bgr)

        if cv2.waitKey(1) & 0xFF == ord('q'):
            break
finally:
    picam2.stop()
    cv2.destroyAllWindows()
